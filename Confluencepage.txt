from sqlalchemy import (
    Column,
    String,
    Integer,
    DateTime,
    Text,
    UniqueConstraint,
    inspect,
)

from sqlalchemy.orm import declarative_base

from db import engine


Base = declarative_base()


class ACLPDocumentMetadata(Base):
    __tablename__ = "ACLP_DOCUMENT_METADATA"

    # Technical primary key
    DOCUMENT_ID = Column(
        Integer,
        primary_key=True,
        autoincrement=True
    )

    # Document fields
    WORKBOOK_NAME = Column(
        String(1000),
        nullable=False
    )

    DOCUMENT_STATUS = Column(
        String(50),
        nullable=False
    )

    DOCUMENT_ERROR = Column(
        Text,
        nullable=True
    )

    SHEET_NAME = Column(
        String(500),
        nullable=True
    )

    HEADER_ROW = Column(
        Integer,
        nullable=True
    )

    # JSON stored as text in Azure SQL
    MATCHING_COLUMNS = Column(
        Text,
        nullable=True
    )

    MATCHING_ROWS = Column(
        Text,
        nullable=True
    )

    # Extracted relationship information
    ENGAGEMENT_ID = Column(
        String(100),
        nullable=True
    )

    SUPPLIER_NAME = Column(
        String(1000),
        nullable=True
    )

    THIRD_PARTY_ID = Column(
        String(100),
        nullable=True
    )

    THIRD_PARTY_NAME = Column(
        String(1000),
        nullable=True
    )

    # Batch / source metadata
    RUN_BATCH_ID = Column(
        String(36),
        nullable=False
    )

    GENERATED_AT = Column(
        DateTime,
        nullable=False
    )

    TARGET_SHEET = Column(
        String(500),
        nullable=False
    )

    CREATED_AT = Column(
        DateTime,
        nullable=False,
        server_default="GETDATE()"
    )

    # Prevent processing the same workbook twice
    # within the same batch.
    __table_args__ = (
        UniqueConstraint(
            "RUN_BATCH_ID",
            "WORKBOOK_NAME",
            name="UQ_ACLP_DOCUMENT_METADATA_BATCH_WORKBOOK"
        ),
    )


def create_table():
    inspector = inspect(engine)

    if inspector.has_table("ACLP_DOCUMENT_METADATA"):
        print(
            "ACLP_DOCUMENT_METADATA table already exists. "
            "No changes made."
        )
        return

    Base.metadata.create_all(
        engine,
        tables=[ACLPDocumentMetadata.__table__]
    )

    print(
        "ACLP_DOCUMENT_METADATA table created successfully."
    )


if __name__ == "__main__":
    create_table()
