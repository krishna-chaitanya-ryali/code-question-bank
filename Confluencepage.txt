from sqlalchemy import text

from database import engine   # use your existing DB connection module


def cleanup_relationship_table():
    delete_query = text("""
        DELETE FROM DIM_TP_ENG_CONTR_BUSAPP_REL
        WHERE engagement_detail_engagement_id IS NULL
    """)

    with engine.begin() as connection:
        result = connection.execute(delete_query)

        print(
            f"Deleted {result.rowcount} records "
            "where engagement_detail_engagement_id is NULL."
        )


if __name__ == "__main__":
    cleanup_relationship_table()
