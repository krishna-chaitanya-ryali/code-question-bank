generic_loader.py


"""
Generic Excel -> Azure SQL loader.

The loader is configuration-driven:
- Input Excel file
- Excel sheet
- Target SQL table
- Optional record-level filter
- Optional source-to-target column mapping

The target SQL schema is read dynamically from the database.
"""

from __future__ import annotations

from pathlib import Path
from typing import Any

import pandas as pd
import yaml
from sqlalchemy import inspect
from sqlalchemy.sql.sqltypes import (
    Boolean,
    Date,
    DateTime,
    Float,
    Integer,
    Numeric,
)

from db import engine


CONFIG_FILE = "loader_config.yaml"


def load_config(config_file: str = CONFIG_FILE) -> dict[str, Any]:
    """Read YAML configuration."""

    with open(config_file, "r", encoding="utf-8") as file:
        config = yaml.safe_load(file)

    if not isinstance(config, dict):
        raise ValueError("Configuration file must contain a YAML mapping.")

    return config


def validate_config(config: dict[str, Any]) -> None:
    """Validate mandatory configuration fields."""

    required_fields = [
        "input_file",
        "sheet_name",
        "target_table",
    ]

    missing = [
        field
        for field in required_fields
        if not config.get(field)
    ]

    if missing:
        raise ValueError(
            f"Missing required configuration fields: {', '.join(missing)}"
        )


def read_excel_data(
    input_file: str,
    sheet_name: str,
) -> pd.DataFrame:
    """Read the configured Excel sheet into a DataFrame."""

    file_path = Path(input_file)

    if not file_path.exists():
        raise FileNotFoundError(
            f"Input Excel file not found: {file_path.resolve()}"
        )

    print(f"Reading Excel file: {file_path}")
    print(f"Reading sheet: {sheet_name}")

    df = pd.read_excel(
        file_path,
        sheet_name=sheet_name,
    )

    print(f"Rows read from Excel: {len(df)}")
    print(f"Columns read from Excel: {len(df.columns)}")

    return df


def apply_column_mapping(
    df: pd.DataFrame,
    column_mapping: dict[str, str] | None,
) -> pd.DataFrame:
    """
    Rename source Excel columns to target SQL columns.

    Example:

    column_mapping:
        SOURCE_COLUMN: target_column
    """

    if not column_mapping:
        return df

    print("Applying column mapping...")

    missing_source_columns = [
        source_column
        for source_column in column_mapping
        if source_column not in df.columns
    ]

    if missing_source_columns:
        raise ValueError(
            "The following source columns are missing from Excel: "
            + ", ".join(missing_source_columns)
        )

    return df.rename(columns=column_mapping)


def apply_record_filter(
    df: pd.DataFrame,
    record_filter: dict[str, Any] | None,
) -> pd.DataFrame:
    """Apply optional record-level filtering."""

    if not record_filter:
        print("No record-level filter configured. All rows will be processed.")
        return df

    key_column = record_filter.get("key_column")
    values = record_filter.get("values")

    if not key_column:
        raise ValueError(
            "record_filter.key_column is required when record_filter is used."
        )

    if not values:
        raise ValueError(
            "record_filter.values must contain at least one value."
        )

    if key_column not in df.columns:
        raise ValueError(
            f"Filter key column '{key_column}' was not found in Excel data."
        )

    # Convert both sides to string so that values such as numeric IDs
    # are compared consistently with Excel values.
    filter_values = {str(value).strip() for value in values}

    original_count = len(df)

    df = df[
        df[key_column]
        .astype("string")
        .str.strip()
        .isin(filter_values)
    ].copy()

    print(f"Filter key column: {key_column}")
    print(f"Requested filter values: {len(filter_values)}")
    print(f"Rows before filtering: {original_count}")
    print(f"Rows after filtering: {len(df)}")

    return df


def read_target_schema(target_table: str) -> dict[str, Any]:
    """Read target table columns and SQL data types from the database."""

    inspector = inspect(engine)

    if not inspector.has_table(target_table):
        raise RuntimeError(
            f"Target table '{target_table}' does not exist in the database."
        )

    columns = inspector.get_columns(target_table)

    schema = {
        column["name"]: column["type"]
        for column in columns
    }

    print(f"\nTarget table: {target_table}")
    print("Target database schema:")

    for column_name, column_type in schema.items():
        print(f"  {column_name}: {column_type}")

    return schema


def validate_columns(
    df: pd.DataFrame,
    target_schema: dict[str, Any],
) -> None:
    """Validate that DataFrame columns exist in the target table."""

    source_columns = set(df.columns)
    target_columns = set(target_schema.keys())

    missing_in_excel = target_columns - source_columns

    if missing_in_excel:
        raise ValueError(
            "Target table columns missing from Excel/DataFrame: "
            + ", ".join(sorted(missing_in_excel))
        )

    extra_columns = source_columns - target_columns

    if extra_columns:
        print(
            "\nWarning: These Excel columns are not present in the target table "
            "and will be ignored:"
        )

        for column in sorted(extra_columns):
            print(f"  {column}")


def convert_column_to_sql_type(
    series: pd.Series,
    sql_type: Any,
) -> pd.Series:
    """Convert a Pandas series according to the target SQL type."""

    # Integer
    if isinstance(sql_type, Integer):
        return pd.to_numeric(
            series,
            errors="coerce",
        ).astype("Int64")

    # Float / Numeric
    if isinstance(sql_type, (Float, Numeric)):
        return pd.to_numeric(
            series,
            errors="coerce",
        )

    # Boolean
    if isinstance(sql_type, Boolean):
        if series.dtype == bool:
            return series

        return (
            series.astype("string")
            .str.strip()
            .str.upper()
            .map(
                {
                    "TRUE": True,
                    "FALSE": False,
                    "1": True,
                    "0": False,
                    "YES": True,
                    "NO": False,
                }
            )
        )

    # DateTime
    if isinstance(sql_type, DateTime):
        return pd.to_datetime(
            series,
            errors="coerce",
        )

    # Date
    if isinstance(sql_type, Date):
        return pd.to_datetime(
            series,
            errors="coerce",
        ).dt.date

    # Everything else is treated as string/text.
    return series.astype("string")


def apply_target_schema(
    df: pd.DataFrame,
    target_schema: dict[str, Any],
) -> pd.DataFrame:
    """Apply SQL target data types to DataFrame columns."""

    print("\nConverting DataFrame according to target SQL schema...")

    for column_name, sql_type in target_schema.items():

        if column_name not in df.columns:
            continue

        print(f"Converting: {column_name} -> {sql_type}")

        df[column_name] = convert_column_to_sql_type(
            df[column_name],
            sql_type,
        )

    return df


def prepare_for_sql(
    df: pd.DataFrame,
    target_schema: dict[str, Any],
) -> pd.DataFrame:
    """Prepare DataFrame for SQL insertion."""

    # Only send columns that actually exist in target table.
    target_columns = [
        column_name
        for column_name in target_schema
        if column_name in df.columns
    ]

    df = df[target_columns].copy()

    # Pandas NaN / NaT must become Python None for SQL NULL.
    df = df.astype(object).where(
        pd.notna(df),
        None,
    )

    return df


def load_to_sql(
    df: pd.DataFrame,
    target_table: str,
) -> None:
    """Insert prepared DataFrame into existing SQL table."""

    if df.empty:
        print("\nNo rows to load.")
        return

    print(
        f"\nLoading {len(df)} rows into "
        f"{target_table}..."
    )

    df.to_sql(
        name=target_table,
        con=engine,
        if_exists="append",
        index=False,
        chunksize=100,
    )

    print("Data loaded successfully.")


def main() -> int:
    """Execute the complete generic loading process."""

    print("=" * 70)
    print("WS5 GENERIC EXCEL -> SQL LOADER")
    print("=" * 70)

    config = load_config()
    validate_config(config)

    input_file = config["input_file"]
    sheet_name = config["sheet_name"]
    target_table = config["target_table"]

    # 1. Read Excel.
    df = read_excel_data(
        input_file=input_file,
        sheet_name=sheet_name,
    )

    # 2. Optional source -> target column mapping.
    df = apply_column_mapping(
        df,
        config.get("column_mapping"),
    )

    # 3. Optional record filtering.
    df = apply_record_filter(
        df,
        config.get("record_filter"),
    )

    if df.empty:
        print("\nNo matching records found.")
        return 0

    # 4. Read target SQL schema.
    target_schema = read_target_schema(
        target_table,
    )

    # 5. Validate columns.
    validate_columns(
        df,
        target_schema,
    )

    # 6. Apply SQL types.
    df = apply_target_schema(
        df,
        target_schema,
    )

    # 7. Prepare NULLs and target columns.
    df = prepare_for_sql(
        df,
        target_schema,
    )

    print("\nFinal DataFrame:")
    print(df.head())

    print(f"\nFinal row count: {len(df)}")
    print(f"Final column count: {len(df.columns)}")

    # 8. Load into SQL.
    load_to_sql(
        df,
        target_table,
    )

    print("\n" + "=" * 70)
    print("LOAD COMPLETED")
    print("=" * 70)

    return 0


if __name__ == "__main__":
    raise SystemExit(main())
