db.py

from sqlalchemy import create_engine

SERVER = "proc-prepoc-ws5-sql-01.database.windows.net"

DATABASE = "proc-prepoc-ws5-sql-db-01"

connection_string = (
    "mssql+pyodbc://@"
    f"{SERVER}/{DATABASE}"
    "?driver=ODBC+Driver+18+for+SQL+Server"
    "&Authentication=ActiveDirectoryInteractive"
)

engine = create_engine(
    connection_string,
    echo=True,
    fast_executemany=True
)


model.py


from sqlalchemy.orm import declarative_base
from sqlalchemy import Column,String

Base = declarative_base()

class Engagement(Base):

    __tablename__="ENGAGEMENT"

    engagement_id = Column(String(100),primary_key=True)

    supplier_key_text = Column(String(100))

    engagement_name = Column(String(500))

    engagement_desc = Column(String(1000))

    engagement_status = Column(String(100))

    category_name = Column(String(200))

    important_third_party_service = Column(String(20))

    important_business_service = Column(String(300))

    materiality_assessment = Column(String(200))

    inherent_rating = Column(String(100))

    application_instance_id = Column(String(100))

    global_business_name = Column(String(300))

    type = Column(String(100))

    tpem_employee_id = Column(String(100))

    tpem_email = Column(String(200))

    engagement_risk_owner_employee_id = Column(String(100))

    engagement_risk_owner_email = Column(String(200))

load_engagement

import pandas as pd

from sqlalchemy.orm import Session

from db import engine

from models import Engagement

df = pd.read_excel(
    "engagement_info.xlsx",
    sheet_name="Engagement Info"
)

session = Session(engine)

for _,row in df.iterrows():

    engagement = Engagement(

        engagement_id=row["ENGAGEMENT_DETAIL_ENGAGEMENT_ID"],

        supplier_key_text=row["SUPPLIER_KEY_TEXT"],

        engagement_name=row["ENGAGEMENT_DETAIL_ENGAGEMENT_NAME"],

        engagement_desc=row["ENGAGEMENT_DETAIL_DESC"],

        engagement_status=row["ENGAGEMENT_DETAIL_ENGAGEMENT_STATUS_NAME"],

        category_name=row["ENGAGEMENT_DETAIL_CATEGORY_NAME"],

        important_third_party_service=row["IMPORTANT_THIRD_PARTY_SERVICE_IND"],

        important_business_service=row["ENGAGEMENT_DETAIL_IMPORTANT_BUSINESS_SERVICE_NAME"],

        materiality_assessment=row["ENGAGEMENT_DETAIL_MATERIALITY_ASSESSMENT_OUTCOME_TEXT"],

        inherent_rating=row["ENGAGEMENT_DETAIL_INHERENT_RATING_TEXT"],

        application_instance_id=row["ENGAGEMENT_DETAIL_APPLICATION_INSTANCE_ID"],

        global_business_name=row["SUPPLIER_DNB_GLOBAL_ULTIMATE_BUSINESS_NAME"],

        type=row["type"],

        tpem_employee_id=row["TPEM_EMPLOYEE_ID"],

        tpem_email=row["TPEM_EMAIL_ID"],

        engagement_risk_owner_employee_id=row["ENGAGEMENT_RISK_OWNER_EMPLOYEE_ID"],

        engagement_risk_owner_email=row["ENGAGEMENT_RISK_OWNER_EMAIL_ID"]
    )

    session.merge(engagement)

session.commit()

session.close()

print("Data Loaded Successfully")
