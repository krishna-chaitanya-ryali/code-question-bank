from sqlalchemy import text
from db import engine

TABLE_NAME = "BSO_IT_SHEETS"


def create_table():
    ddl = f"""
    IF OBJECT_ID('{TABLE_NAME}', 'U') IS NULL
    BEGIN
        CREATE TABLE {TABLE_NAME} (
            IMPORTANT_PROCESS VARCHAR(100) NULL,
            PROCESS_HANDOFF VARCHAR(100) NULL,
            BUSINESS_APPLICATION_ID VARCHAR(100) NULL,
            BUSINESS_APPLICATION_NAME VARCHAR(500) NULL,
            EUC BIT NULL,
            DOMINO BIT NULL,
            HAS_HAD_AN_INCIDENT VARCHAR(20) NULL,
            APP_STATUS VARCHAR(100) NULL,
            BIA_OVERALL_SCORE VARCHAR(100) NULL,
            TRMF_RTO DECIMAL(10,4) NULL,
            TRMF_RPO DECIMAL(10,4) NULL,
            TRMF_AVAILABILITY_CORE DECIMAL(10,4) NULL,
            TRMF_AVAILABILITY_ENHANCED DECIMAL(10,4) NULL,
            TRMF_ADDITIONAL_CAPACITY DECIMAL(10,4) NULL,
            OWNER VARCHAR(300) NULL,
            DISASTER_RECOVERY_STATUS VARCHAR(100) NULL,
            STATUS VARCHAR(100) NULL,
            RTO INT NULL,
            RPO VARCHAR(100) NULL,
            RTA VARCHAR(50) NULL,
            DC_RTO_MINS INT NULL,
            DC_RPO_MINS INT NULL,
            LAST_COMPLETION DATE NULL,
            TESTING_DUE DATE NULL,
            SCHEDULED_TEST DATE NULL,
            ASSOCIATED_IBS INT NULL,
            LAST_12_MONTH_OUTAGE_COUNT INT NULL
        );
    END
    """

    with engine.begin() as conn:
        conn.execute(text(ddl))

    print(f"Table {TABLE_NAME} created successfully (or already exists).")


if __name__ == "__main__":
    create_table()


load_bso_it_sheets.py

import pandas as pd
from sqlalchemy import text
from db import engine

TABLE_NAME = "BSO_IT_SHEETS"
EXCEL_FILE = "Data workbook.xlsx"
SHEET_NAME = "BSO_IT_SHEETS"


SOURCE_TO_TARGET = {
    "Important Process": "IMPORTANT_PROCESS",
    "Process Handoff": "PROCESS_HANDOFF",
    "Business App ID": "BUSINESS_APPLICATION_ID",
    "Business App Name": "BUSINESS_APPLICATION_NAME",
    "EUC": "EUC",
    "Domino": "DOMINO",
    "Has had an incident?": "HAS_HAD_AN_INCIDENT",
    "App Status": "APP_STATUS",
    "BIA Overall Score": "BIA_OVERALL_SCORE",
    "TRMF RTO": "TRMF_RTO",
    "TRMF RPO": "TRMF_RPO",
    "TRMF Availability Core": "TRMF_AVAILABILITY_CORE",
    "TRMF Availability Enhanced": "TRMF_AVAILABILITY_ENHANCED",
    "TRMF Additional Capacity": "TRMF_ADDITIONAL_CAPACITY",
    "Owner": "OWNER",
    "Disaster Recovery Status": "DISASTER_RECOVERY_STATUS",
    "Status": "STATUS",
    "RTO": "RTO",
    "RPO": "RPO",
    "RTA": "RTA",
    "DC RTO (Mins)": "DC_RTO_MINS",
    "DC RPO (Mins)": "DC_RPO_MINS",
    "Last Completion": "LAST_COMPLETION",
    "Testing Due": "TESTING_DUE",
    "Scheduled Test": "SCHEDULED_TEST",
    "Associated IBS": "ASSOCIATED_IBS",
    "Last 12 month outage count": "LAST_12_MONTH_OUTAGE_COUNT",
}


def load_data():

    print(f"Reading sheet '{SHEET_NAME}' from '{EXCEL_FILE}'...")

    df = pd.read_excel(
        EXCEL_FILE,
        sheet_name=SHEET_NAME,
        dtype=str
    )

    # Remove accidental spaces from column headers
    df.columns = df.columns.str.strip()

    # Validate expected source columns
    expected_columns = list(SOURCE_TO_TARGET.keys())

    missing_columns = [
        column
        for column in expected_columns
        if column not in df.columns
    ]

    if missing_columns:
        raise ValueError(
            f"Missing expected columns in sheet '{SHEET_NAME}': "
            f"{missing_columns}"
        )

    # Keep only required columns
    df = df[expected_columns]

    # Rename source columns to target database columns
    df = df.rename(columns=SOURCE_TO_TARGET)

    # Convert blank/whitespace values to NULL
    df = df.replace(r"^\s*$", pd.NA, regex=True)

    # Convert BIT columns
    bit_columns = [
        "EUC",
        "DOMINO",
    ]

    for column in bit_columns:
        df[column] = (
            df[column]
            .astype("string")
            .str.strip()
            .str.upper()
            .map({
                "TRUE": 1,
                "FALSE": 0,
                "1": 1,
                "0": 0,
            })
        )

    # Convert decimal/numeric columns
    decimal_columns = [
        "TRMF_RTO",
        "TRMF_RPO",
        "TRMF_AVAILABILITY_CORE",
        "TRMF_AVAILABILITY_ENHANCED",
        "TRMF_ADDITIONAL_CAPACITY",
    ]

    for column in decimal_columns:
        df[column] = pd.to_numeric(
            df[column],
            errors="coerce"
        )

    # Convert integer columns
    integer_columns = [
        "RTO",
        "DC_RTO_MINS",
        "DC_RPO_MINS",
        "ASSOCIATED_IBS",
        "LAST_12_MONTH_OUTAGE_COUNT",
    ]

    for column in integer_columns:
        df[column] = pd.to_numeric(
            df[column],
            errors="coerce"
        )

    # Convert date columns
    date_columns = [
        "LAST_COMPLETION",
        "TESTING_DUE",
        "SCHEDULED_TEST",
    ]

    for column in date_columns:
        df[column] = pd.to_datetime(
            df[column],
            errors="coerce",
            dayfirst=True
        ).dt.date

    print(f"Rows read from Excel: {len(df):,}")

    # Clear existing data before reload
    with engine.begin() as conn:
        conn.execute(text(f"DELETE FROM {TABLE_NAME}"))

    print(f"Existing data cleared from {TABLE_NAME}.")

    # Load data into Azure SQL
    df.to_sql(
        TABLE_NAME,
        con=engine,
        if_exists="append",
        index=False,
        chunksize=500,
        method=None
    )

    print(
        f"Successfully loaded {len(df):,} rows "
        f"into {TABLE_NAME}."
    )


if __name__ == "__main__":
    load_data()
