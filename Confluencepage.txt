rename_bs4_to_bso_technology_thresholds.py



from sqlalchemy import text

from db import engine


OLD_TABLE_NAME = "BSO_TECHNOLOGY_THRESHOLDS"
NEW_TABLE_NAME = "BS4_TECHNOLOGY_THRESHOLDS"


def rename_table():

    with engine.begin() as connection:

        # Drop the existing BSO table
        connection.execute(
            text(f"""
                IF OBJECT_ID('dbo.{OLD_TABLE_NAME}', 'U') IS NOT NULL
                BEGIN
                    DROP TABLE dbo.{OLD_TABLE_NAME};
                END
            """)
        )

        # Rename the newly created BS4 table to BSO
        connection.execute(
            text(f"""
                IF OBJECT_ID('dbo.{NEW_TABLE_NAME}', 'U') IS NOT NULL
                BEGIN
                    EXEC sp_rename
                        'dbo.{NEW_TABLE_NAME}',
                        '{OLD_TABLE_NAME}';
                END
            """)
        )

    print(
        f"Table '{NEW_TABLE_NAME}' renamed to '{OLD_TABLE_NAME}' successfully."
    )


if __name__ == "__main__":
    rename_table()
