from sqlalchemy.orm import declarative_base, relationship
from sqlalchemy import Column, String, ForeignKey

Base = declarative_base()

# ---------------------------------------
# Existing Engagement model
# ---------------------------------------

class Engagement(Base):

    __tablename__ = "ENGAGEMENT"

    engagement_id = Column(String(100), primary_key=True)

    supplier_key_text = Column(String(100))
    engagement_name = Column(String(500))
    engagement_desc = Column(String(1000))
    engagement_status = Column(String(100))
    category_name = Column(String(200))
    important_third_party_service = Column(String(20))
    important_business_service = Column(String(300))
    materiality_assessment = Column(String(200))
    inherent_rating = Column(String(100))
    application_instance_id = Column(String(100))
    global_business_name = Column(String(300))
    type = Column(String(100))
    tpem_employee_id = Column(String(100))
    tpem_email = Column(String(200))
    engagement_risk_owner_employee_id = Column(String(100))
    engagement_risk_owner_email = Column(String(200))

    contracts = relationship("Contract", back_populates="engagement")
	

class ThirdParty(Base):

    __tablename__ = "THIRD_PARTY"

    third_party_id = Column(String(100), primary_key=True)

    global_tp_id = Column(String(100))

    name = Column(String(300))

    status = Column(String(100))

    contracts = relationship("Contract", back_populates="third_party")
	

class Contract(Base):

    __tablename__ = "CONTRACT"

    contract_ref_num = Column(String(100), primary_key=True)

    engagement_id = Column(
        String(100),
        ForeignKey("ENGAGEMENT.engagement_id"),
        nullable=False
    )

    third_party_id = Column(
        String(100),
        ForeignKey("THIRD_PARTY.third_party_id"),
        nullable=False
    )

    contract_desc = Column(String(1000))

    contract_status = Column(String(100))

    contract_type = Column(String(200))

    engagement = relationship("Engagement", back_populates="contracts")

    third_party = relationship("ThirdParty", back_populates="contracts")


class BusinessService(Base):

    __tablename__ = "BUS_SERVICE"

    unique_id = Column(String(100), primary_key=True)

    service_name = Column(String(500))

    service_description = Column(String(2000))

    owner = Column(String(300))

    status = Column(String(100))
	

class GtpEngContractRelation(Base):

    __tablename__ = "GTP_ENG_CONTR_REL"

    third_party_id = Column(
        String(100),
        ForeignKey("THIRD_PARTY.third_party_id"),
        primary_key=True
    )

    engagement_id = Column(
        String(100),
        ForeignKey("ENGAGEMENT.engagement_id"),
        primary_key=True
    )

    contract_ref_num = Column(
        String(100),
        ForeignKey("CONTRACT.contract_ref_num"),
        primary_key=True
    )

    global_tp_id = Column(String(100))

    relationship_type = Column(String(100))

    remarks = Column(String(500))
