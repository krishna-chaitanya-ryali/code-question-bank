SELECT
    BUS_ARCH_OBJECT_TYPE_CODE,
    BUS_ARCH_OBJECT_NAME,
    ENGAGEMENT_ID,
    BUSINESS_APPLICATION_ID,
    COUNT(*) AS RECORD_COUNT
FROM BUS_SERV_AWARE_ENGAGEMENT
GROUP BY
    BUS_ARCH_OBJECT_TYPE_CODE,
    BUS_ARCH_OBJECT_NAME,
    ENGAGEMENT_ID,
    BUSINESS_APPLICATION_ID
HAVING COUNT(*) > 1;
