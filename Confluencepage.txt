SELECT
    SUM(CASE WHEN BUS_ARCH_OBJECT_TYPE_CODE IS NULL THEN 1 ELSE 0 END) AS TYPE_NULLS,
    SUM(CASE WHEN BUS_ARCH_OBJECT_NAME IS NULL THEN 1 ELSE 0 END) AS NAME_NULLS,
    SUM(CASE WHEN ENGAGEMENT_ID IS NULL THEN 1 ELSE 0 END) AS ENGAGEMENT_NULLS,
    SUM(CASE WHEN BUSINESS_APPLICATION_ID IS NULL THEN 1 ELSE 0 END) AS APPLICATION_NULLS
FROM BUS_SERV_AWARE_ENGAGEMENT;
