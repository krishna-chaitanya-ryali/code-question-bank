import pandas as pd
from sqlalchemy import text
from db import engine


TABLE_NAME = "IBS_PREMISES_DETAILS"
SHEET_NAME = "IBS_PREMISES_DETAILS"

EXCEL_FILE = "Data workbook.xlsx"


COLUMNS = [
    "BUS_ARCH_OBJECT_TYPE_CODE",
    "BUS_ARCH_OBJECT_ID",
    "BUS_ARCH_OBJECT_NAME",
    "BUS_ARCH_OWNER_NAME",
    "BUS_ARCH_OWNER_EMAIL_TEXT",
    "BUS_ARCH_LEGAL_ENTITY_ID",
    "BUS_ARCH_LEGAL_ENTITY_NAME",
    "BUS_ARCH_LEGAL_ENTITY_CODE",
    "BUS_ARCH_LEGAL_ENTITY_REGION_CODE",
    "SERVICE_IMPORTANCE_IND",
    "SERVICE_GROUP_IMPORTANT_IND",
    "PROPERTY_ID",
    "PROPERTY_NAME",
    "PROPERTY_COUNTRY_TERRITORY_ISO_CODE",
    "PROPERTY_COUNTRY_TERRITORY_ISO_NAME",
    "PROPERTY_REGION_CODE",
    "PROPERTY_REGION_NAME",
    "PROPERTY_CITY_NAME",
    "PROPERTY_ADDRESS_LINE1_TEXT",
    "PROPERTY_ADDRESS_LINE2_TEXT",
    "PROPERTY_DETAIL_PROPERTY_STATUS_NAME",
    "PROPERTY_DETAIL_PROPERTY_TYPE_NAME",
    "REFERENCE_MAPPING_DIRECT_IND",
    "REFERENCE_MAPPING_HANDOFF_IND",
    "ASSOCIATION_IMPORTANT_IND",
    "REPORT_DATE",
]


def convert_boolean(value):
    if pd.isna(value):
        return None

    if isinstance(value, bool):
        return value

    value = str(value).strip().upper()

    if value in ("TRUE", "1", "YES", "Y"):
        return True

    if value in ("FALSE", "0", "NO", "N"):
        return False

    return None


def load_data():

    print(f"Reading sheet '{SHEET_NAME}' from '{EXCEL_FILE}'...")

    df = pd.read_excel(
        EXCEL_FILE,
        sheet_name=SHEET_NAME,
        dtype=str
    )

    print(f"Rows read from Excel: {len(df)}")

    # Normalize column names
    df.columns = [str(col).strip() for col in df.columns]

    # Validate expected columns
    missing_columns = [
        column for column in COLUMNS
        if column not in df.columns
    ]

    if missing_columns:
        raise ValueError(
            f"Missing columns in Excel sheet: {missing_columns}"
        )

    # Keep only the expected columns and preserve their order
    df = df[COLUMNS].copy()

    # Convert blank strings to None
    df = df.replace(r"^\s*$", None, regex=True)

    # Convert BIT columns
    bit_columns = [
        "SERVICE_IMPORTANCE_IND",
        "SERVICE_GROUP_IMPORTANT_IND",
        "REFERENCE_MAPPING_DIRECT_IND",
        "REFERENCE_MAPPING_HANDOFF_IND",
        "ASSOCIATION_IMPORTANT_IND",
    ]

    for column in bit_columns:
        df[column] = df[column].apply(convert_boolean)

    # Convert REPORT_DATE to SQL DATE
    df["REPORT_DATE"] = pd.to_datetime(
        df["REPORT_DATE"],
        errors="coerce"
    ).dt.date

    # Load into SQL Server
    print(f"Loading {len(df)} rows into '{TABLE_NAME}'...")

    with engine.begin() as connection:

        # Clear previous snapshot data.
        # This makes the script safe to re-run without duplicating rows.
        connection.execute(
            text(f"DELETE FROM {TABLE_NAME}")
        )

        df.to_sql(
            TABLE_NAME,
            con=connection,
            if_exists="append",
            index=False,
            chunksize=100,
            method=None
        )

    print(
        f"Successfully loaded {len(df)} rows into '{TABLE_NAME}'."
    )


if __name__ == "__main__":
    load_data()
