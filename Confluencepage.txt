SELECT
    ARIS_SERVICE_ID,
    COUNT(*) AS RECORD_COUNT
FROM SERVICE_TOLERANCES
WHERE ARIS_SERVICE_ID IS NOT NULL
GROUP BY ARIS_SERVICE_ID
HAVING COUNT(*) > 1
ORDER BY RECORD_COUNT DESC;



SELECT *
FROM SERVICE_TOLERANCES
WHERE ARIS_SERVICE_ID IN (
    SELECT ARIS_SERVICE_ID
    FROM SERVICE_TOLERANCES
    WHERE ARIS_SERVICE_ID IS NOT NULL
    GROUP BY ARIS_SERVICE_ID
    HAVING COUNT(*) > 1
)
ORDER BY ARIS_SERVICE_ID;




SELECT COUNT(*) AS DUPLICATE_SERVICE_ID_COUNT
FROM (
    SELECT ARIS_SERVICE_ID
    FROM SERVICE_TOLERANCES
    WHERE ARIS_SERVICE_ID IS NOT NULL
    GROUP BY ARIS_SERVICE_ID
    HAVING COUNT(*) > 1
) AS D;
