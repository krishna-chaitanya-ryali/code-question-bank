SELECT
    BUSINESS_SERVICE_TYPE,
    LEGAL_ENTITY,
    DIRECT_INDIRECT_MAPPING,
    SOURCE,
    PROCESS_HANDOFF,
    ENGAGEMENT_ID,
    APP_ID,
    COUNT(*) AS RECORD_COUNT
FROM IMPORTANT_BUSINESS_SERVICE_DETAILS
GROUP BY
    BUSINESS_SERVICE_TYPE,
    LEGAL_ENTITY,
    DIRECT_INDIRECT_MAPPING,
    SOURCE,
    PROCESS_HANDOFF,
    ENGAGEMENT_ID,
    APP_ID
HAVING COUNT(*) > 1;
