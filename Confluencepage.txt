from sqlalchemy import text

from db import engine


CONSTRAINT_NAME = "UQ_ACLP_DOCUMENT_METADATA_BATCH_WORKBOOK"


def remove_unique_constraint():
    sql = text("""
        ALTER TABLE ACLP_DOCUMENT_METADATA
        DROP CONSTRAINT UQ_ACLP_DOCUMENT_METADATA_BATCH_WORKBOOK
    """)

    with engine.begin() as connection:
        connection.execute(sql)

    print(
        "Unique constraint "
        "UQ_ACLP_DOCUMENT_METADATA_BATCH_WORKBOOK "
        "removed successfully."
    )


if __name__ == "__main__":
    remove_unique_constraint()
