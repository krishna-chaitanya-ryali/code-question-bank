SELECT
    SUPPLIER_KEY_TEXT,
    ENGAGEMENT_DETAIL_ENGAGEMENT_ID,
    CONTRACTS_DETAIL_CONTRACT_NUMBER,
    BUSINESS_APPLICATION_ID,
    SUPPLIER_DNB_GLOBAL_ULTIMATE_BUSINESS_NAME,
    SUPPLIER_NORMALISED_BUSINESS_NAME,
    SUPPLIER_DNB_GLOBAL_ULTIMATE_BUSINESS_NAME_1,
    COUNT(*) AS RECORD_COUNT
FROM DIM_TP_ENG_CONTR_BUSAPP_REL
GROUP BY
    SUPPLIER_KEY_TEXT,
    ENGAGEMENT_DETAIL_ENGAGEMENT_ID,
    CONTRACTS_DETAIL_CONTRACT_NUMBER,
    BUSINESS_APPLICATION_ID,
    SUPPLIER_DNB_GLOBAL_ULTIMATE_BUSINESS_NAME,
    SUPPLIER_NORMALISED_BUSINESS_NAME,
    SUPPLIER_DNB_GLOBAL_ULTIMATE_BUSINESS_NAME_1
HAVING COUNT(*) > 1;
