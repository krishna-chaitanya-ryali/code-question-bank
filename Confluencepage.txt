class Contract(Base):
    """Contract records that join an engagement with a third party."""

    __tablename__ = "CONTRACT"

    __table_args__ = (
        # Required so GTP_ENG_CONTR_REL can reference the
        # contract + engagement + third-party combination.
        UniqueConstraint(
            "contract_ref_num",
            "engagement_id",
            "third_party_id",
            name="UQ_CONTRACT_REF_ENG_TP",
        ),
    )

    # Existing primary key from the original POC design.
    #
    # Do NOT rename this to CONTRACTS_DETAIL_CONTRACT_NUMBER yet.
    # We need to verify that both represent the same business key.
    contract_ref_num = Column(
        String(100),
        primary_key=True,
    )

    # Foreign key:
    # CONTRACT.engagement_id
    #       -> ENGAGEMENT.engagement_id
    engagement_id = Column(
        String(100),
        ForeignKey("ENGAGEMENT.engagement_id"),
        nullable=False,
    )

    # Foreign key:
    # CONTRACT.third_party_id
    #       -> THIRD_PARTY.third_party_id
    third_party_id = Column(
        String(100),
        ForeignKey("THIRD_PARTY.third_party_id"),
        nullable=False,
    )

    # ------------------------------------------------------------------
    # Existing POC columns
    # ------------------------------------------------------------------

    # Existing contract description.
    contract_desc = Column(String(1000))

    contract_status = Column(String(100))

    contract_type = Column(String(200))

    # ------------------------------------------------------------------
    # Additional source fields
    # ------------------------------------------------------------------

    # Source: SUPPLIER_KEY_TEXT
    supplier_key_text = Column(String(100))

    # Source: SUPPLIER_DNB_GLOBAL_ULTIMATE_BUSINESS_NAME
    supplier_global_ultimate_business_name = Column(String(300))

    # Source: ENGAGEMENT_DETAIL_ENGAGEMENT_NAME
    engagement_name = Column(String(500))

    # Source: CONTRACTS_DETAIL_CONTRACT_ID
    #
    # Kept separate from contract_ref_num because we have not yet
    # confirmed that CONTRACT_ID and CONTRACT_NUMBER are the same thing.
    contract_id = Column(String(100))

    # Source: CONTRACTS_DETAIL_CONTRACT_NAME
    contract_name = Column(String(500))

    # Source: CONTRACTS_DETAIL_REGIONAL_COVERAGE_TEXT
    regional_coverage_text = Column(String(500))

    # Source: CONTRACTS_DETAIL_CONTRACT_STATUS_CODE
    contract_status_code = Column(String(100))

    # Source: CONTRACTS_DETAIL_TERMINATION_NOTICE_TEXT
    termination_notice_text = Column(String(500))

    # Source: CONTRACTS_DETAIL_TERMINATION_NOTICE_LENGTH_TEXT
    termination_notice_length_text = Column(String(100))

    # Source: CONTRACTS_DETAIL_TERMINATED_TEXT
    terminated_text = Column(String(50))

    # Source: CONTRACTS_DETAIL_TERMINATION_REASON_CODE
    termination_reason_code = Column(String(100))

    # Source: CONTRACTS_DETAIL_SUPPLIER_NOTICE_PERIOD_DAYS_NUM
    supplier_notice_period_days_num = Column(String(50))

    # Source: CONTRACTS_DETAIL_TERMINATION_CONVENIENCE_RIGHT_IND
    termination_convenience_right_ind = Column(String(50))

    # ------------------------------------------------------------------
    # ORM relationships
    # ------------------------------------------------------------------

    # Parent engagement.
    engagement = relationship(
        "Engagement",
        back_populates="contracts",
    )

    # Parent third party.
    third_party = relationship(
        "ThirdParty",
        back_populates="contracts",
    )

    # Association rows that reference this contract with
    # engagement + third-party IDs.
    gtp_relations = relationship(
        "GtpEngContractRelation",
        back_populates="contract",
        overlaps="gtp_relations,third_party,engagement",
    )


SELECT
    COLUMN_NAME,
    DATA_TYPE,
    CHARACTER_MAXIMUM_LENGTH
FROM INFORMATION_SCHEMA.COLUMNS
WHERE TABLE_NAME = 'CONTRACT'
ORDER BY ORDINAL_POSITION;


SELECT
    tc.CONSTRAINT_NAME,
    tc.CONSTRAINT_TYPE,
    kcu.COLUMN_NAME
FROM INFORMATION_SCHEMA.TABLE_CONSTRAINTS tc
LEFT JOIN INFORMATION_SCHEMA.KEY_COLUMN_USAGE kcu
    ON tc.CONSTRAINT_NAME = kcu.CONSTRAINT_NAME
    AND tc.TABLE_NAME = kcu.TABLE_NAME
WHERE tc.TABLE_NAME = 'CONTRACT'
ORDER BY tc.CONSTRAINT_TYPE, tc.CONSTRAINT_NAME;
