SELECT COUNT(*) AS NULL_THIRD_PARTY_ID_COUNT
FROM THIRD_PARTY
WHERE THIRD_PARTY_ID IS NULL;


SELECT
    THIRD_PARTY_ID,
    COUNT(*) AS RECORD_COUNT
FROM THIRD_PARTY
GROUP BY THIRD_PARTY_ID
HAVING COUNT(*) > 1;

SELECT
    THIRD_PARTY_ID,
    GLOBAL_TP_ID,
    NAME,
    STATUS
FROM THIRD_PARTY
ORDER BY THIRD_PARTY_ID;
