class Contract(Base):
    """Contract records associated with an engagement and third party."""

    __tablename__ = "CONTRACT"

    __table_args__ = (
        UniqueConstraint(
            "contract_ref_num",
            "engagement_id",
            "third_party_id",
            name="UQ_CONTRACT_REF_ENG_TP",
        ),
    )

    # Source: CONTRACTS_DETAIL_CONTRACT_NUMBER
    # Existing PK used by the current model and GTP relationship.
    contract_ref_num = Column(
        String(100),
        primary_key=True,
    )

    # Source: ENGAGEMENT_DETAIL_ENGAGEMENT_ID
    contract_engagement_id = Column(
        String(100),
        ForeignKey("ENGAGEMENT.engagement_id"),
        nullable=False,
    )

    # Source: SUPPLIER_KEY_TEXT
    contract_third_party_id = Column(
        String(100),
        ForeignKey("THIRD_PARTY.third_party_id"),
        nullable=False,
    )

    # Source: CONTRACTS_DETAIL_CONTRACT_ID
    contract_id = Column(String(100))

    # Source: CONTRACTS_DETAIL_CONTRACT_NAME
    contract_name = Column(String(500))

    # Source: CONTRACTS_DETAIL_CONTRACT_DESC
    contract_desc = Column(String(1000))

    # Source: CONTRACTS_DETAIL_REGIONAL_COVERAGE_TEXT
    regional_coverage = Column(String(500))

    # Source: CONTRACTS_DETAIL_CONTRACT_STATUS_CODE
    contract_status = Column(String(100))

    # Source: CONTRACTS_DETAIL_TERMINATION_NOTICE_TEXT
    termination_notice = Column(String(500))

    # Source: CONTRACTS_DETAIL_TERMINATION_NOTICE_LENGTH_TEXT
    termination_notice_length = Column(String(100))

    # Source: CONTRACTS_DETAIL_TERMINATED_TEXT
    terminated = Column(String(20))

    # Source: CONTRACTS_DETAIL_TERMINATION_REASON_CODE
    termination_reason = Column(String(200))

    # Source: CONTRACTS_DETAIL_SUPPLIER_NOTICE_PERIOD_DAYS_NUM
    supplier_notice_period_days = Column(String(50))

    # Source: CONTRACTS_DETAIL_TERMINATION_CONVENIENCE_RIGHT_IND
    termination_convenience_right = Column(String(20))

    engagement = relationship(
        "Engagement",
        back_populates="contracts",
        foreign_keys=[contract_engagement_id],
    )

    third_party = relationship(
        "ThirdParty",
        back_populates="contracts",
        foreign_keys=[contract_third_party_id],
    )

    gtp_relations = relationship(
        "GtpEngContractRelation",
        back_populates="contract",
        overlaps="gtp_relations,third_party,engagement",
    )

"""Recreate the Contract-related tables using the SQLAlchemy connection."""

from sqlalchemy import text
from sqlalchemy.exc import SQLAlchemyError

from db import engine
from models import Base


def main() -> int:
    """Drop and recreate Contract-related tables."""

    try:
        with engine.begin() as conn:

            print("Dropping GTP_ENG_CONTR_REL table...")

            conn.execute(
                text(
                    """
                    IF OBJECT_ID(
                        'dbo.GTP_ENG_CONTR_REL',
                        'U'
                    ) IS NOT NULL
                    DROP TABLE dbo.GTP_ENG_CONTR_REL;
                    """
                )
            )

            print("Dropping CONTRACT table...")

            conn.execute(
                text(
                    """
                    IF OBJECT_ID(
                        'dbo.CONTRACT',
                        'U'
                    ) IS NOT NULL
                    DROP TABLE dbo.CONTRACT;
                    """
                )
            )

        print("Existing Contract tables removed.")

        # Recreate tables from the current SQLAlchemy models.
        Base.metadata.create_all(bind=engine)

        print("Contract-related tables recreated successfully.")

        return 0

    except SQLAlchemyError as exc:
        print("Failed to recreate Contract-related tables.")
        print(f"Error: {exc}")
        return 1


if __name__ == "__main__":
    raise SystemExit(main())


"""Load Contract data for the three in-scope engagements."""

import pandas as pd
from sqlalchemy.orm import Session

from db import engine
from models import Contract


ENGAGEMENT_IDS = {
    "EN-021445",
    "ENG-1162865",
    "ENG-362309",
}


def read_contract_excel(
    file_path: str = "engagement_info.xlsx",
) -> pd.DataFrame:
    """Read Contract worksheet from Excel."""

    return pd.read_excel(
        file_path,
        sheet_name="Contract",
    )


def clean_value(value):
    """Convert pandas NaN values to None."""

    if pd.isna(value):
        return None

    return value


def load_contract_rows(df: pd.DataFrame) -> int:
    """Load Contract rows for the selected engagements."""

    df = df[
        df["ENGAGEMENT_DETAIL_ENGAGEMENT_ID"].isin(
            ENGAGEMENT_IDS
        )
    ].copy()

    print(f"Contract rows selected: {len(df)}")

    session = Session(engine)

    try:

        for _, row in df.iterrows():

            contract = Contract(
                # CONTRACTS_DETAIL_CONTRACT_NUMBER
                contract_ref_num=str(
                    clean_value(
                        row["CONTRACTS_DETAIL_CONTRACT_NUMBER"]
                    )
                ),

                # ENGAGEMENT_DETAIL_ENGAGEMENT_ID
                contract_engagement_id=str(
                    row["ENGAGEMENT_DETAIL_ENGAGEMENT_ID"]
                ),

                # SUPPLIER_KEY_TEXT
                contract_third_party_id=str(
                    row["SUPPLIER_KEY_TEXT"]
                ),

                # CONTRACTS_DETAIL_CONTRACT_ID
                contract_id=clean_value(
                    row["CONTRACTS_DETAIL_CONTRACT_ID"]
                ),

                # CONTRACTS_DETAIL_CONTRACT_NAME
                contract_name=clean_value(
                    row["CONTRACTS_DETAIL_CONTRACT_NAME"]
                ),

                # CONTRACTS_DETAIL_CONTRACT_DESC
                contract_desc=clean_value(
                    row["CONTRACTS_DETAIL_CONTRACT_DESC"]
                ),

                # CONTRACTS_DETAIL_REGIONAL_COVERAGE_TEXT
                regional_coverage=clean_value(
                    row[
                        "CONTRACTS_DETAIL_REGIONAL_COVERAGE_TEXT"
                    ]
                ),

                # CONTRACTS_DETAIL_CONTRACT_STATUS_CODE
                contract_status=clean_value(
                    row[
                        "CONTRACTS_DETAIL_CONTRACT_STATUS_CODE"
                    ]
                ),

                # CONTRACTS_DETAIL_TERMINATION_NOTICE_TEXT
                termination_notice=clean_value(
                    row[
                        "CONTRACTS_DETAIL_TERMINATION_NOTICE_TEXT"
                    ]
                ),

                # CONTRACTS_DETAIL_TERMINATION_NOTICE_LENGTH_TEXT
                termination_notice_length=clean_value(
                    row[
                        "CONTRACTS_DETAIL_TERMINATION_NOTICE_LENGTH_TEXT"
                    ]
                ),

                # CONTRACTS_DETAIL_TERMINATED_TEXT
                terminated=clean_value(
                    row[
                        "CONTRACTS_DETAIL_TERMINATED_TEXT"
                    ]
                ),

                # CONTRACTS_DETAIL_TERMINATION_REASON_CODE
                termination_reason=clean_value(
                    row[
                        "CONTRACTS_DETAIL_TERMINATION_REASON_CODE"
                    ]
                ),

                # CONTRACTS_DETAIL_SUPPLIER_NOTICE_PERIOD_DAYS_NUM
                supplier_notice_period_days=(
                    str(
                        clean_value(
                            row[
                                "CONTRACTS_DETAIL_SUPPLIER_NOTICE_PERIOD_DAYS_NUM"
                            ]
                        )
                    )
                    if clean_value(
                        row[
                            "CONTRACTS_DETAIL_SUPPLIER_NOTICE_PERIOD_DAYS_NUM"
                        ]
                    )
                    is not None
                    else None
                ),

                # CONTRACTS_DETAIL_TERMINATION_CONVENIENCE_RIGHT_IND
                termination_convenience_right=clean_value(
                    row[
                        "CONTRACTS_DETAIL_TERMINATION_CONVENIENCE_RIGHT_IND"
                    ]
                ),
            )

            session.merge(contract)

        session.commit()

        return len(df)

    except Exception:
        session.rollback()
        raise

    finally:
        session.close()


def main() -> int:
    """Execute Contract data load."""

    df = read_contract_excel()

    row_count = load_contract_rows(df)

    print(
        f"Contract data loaded successfully. "
        f"Rows processed: {row_count}"
    )

    return 0


if __name__ == "__main__":
    raise SystemExit(main())
