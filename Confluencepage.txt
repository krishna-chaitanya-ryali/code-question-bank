SELECT
    COUNT(*) AS NULL_COUNT
FROM IMPORTANT_BUSINESS_SERVICE_DETAILS
WHERE IMPORTANT_BUSINESS_SERVICE IS NULL
   OR ENGAGEMENT_ID IS NULL
   OR APP_ID IS NULL;



SELECT
    IMPORTANT_BUSINESS_SERVICE,
    ENGAGEMENT_ID,
    APP_ID,
    COUNT(*) AS RECORD_COUNT
FROM IMPORTANT_BUSINESS_SERVICE_DETAILS
GROUP BY
    IMPORTANT_BUSINESS_SERVICE,
    ENGAGEMENT_ID,
    APP_ID
HAVING COUNT(*) > 1;
