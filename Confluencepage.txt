"""Create DIM_TP_ENG_CONTR_BUSAPP_REL table.

The table stores supplier, engagement, contract and business
application relationship data from the source Excel file.

A technical identity column is used as the primary key because
the source business columns can contain NULLs and duplicate values.
"""

from sqlalchemy import text
from sqlalchemy.exc import SQLAlchemyError

from db import engine
from models import DimTpEngContrBusAppRel


TABLE_NAME = "DIM_TP_ENG_CONTR_BUSAPP_REL"


def main() -> int:
    """Drop and recreate the relationship table."""

    try:
        print(f"Checking existing {TABLE_NAME} table...")

        with engine.begin() as conn:
            conn.execute(
                text(
                    f"""
                    IF OBJECT_ID('{TABLE_NAME}', 'U') IS NOT NULL
                        DROP TABLE {TABLE_NAME};
                    """
                )
            )

        print(f"Existing {TABLE_NAME} table dropped, if it existed.")

        DimTpEngContrBusAppRel.__table__.create(
            bind=engine,
            checkfirst=True,
        )

        print(f"{TABLE_NAME} table created successfully.")
        return 0

    except SQLAlchemyError as exc:
        print(f"Failed to create {TABLE_NAME} table.")
        print(f"Error: {exc}")
        return 1


if __name__ == "__main__":
    raise SystemExit(main())

"""Load supplier, engagement, contract and business application
relationship data from Excel into DIM_TP_ENG_CONTR_BUSAPP_REL.
"""

import pandas as pd
from sqlalchemy import insert
from sqlalchemy.orm import Session

from db import engine
from models import DimTpEngContrBusAppRel


EXCEL_FILE = "engagement_info.xlsx"
SHEET_NAME = "DIM_TP_ENG_CONTR_BUSAPP_REL"


SOURCE_COLUMNS = [
    "SUPPLIER_DNB_GLOBAL_ULTIMATE_BUSINESS_NAME",
    "SUPPLIER_NORMALISED_BUSINESS_NAME",
    "SUPPLIER_KEY_TEXT",
    "ENGAGEMENT_DETAIL_ENGAGEMENT_ID",
    "CONTRACTS_DETAIL_CONTRACT_ID",
    "CONTRACTS_DETAIL_CONTRACT_NUMBER",
    "SUPPLIER_DNB_GLOBAL_ULTIMATE_BUSINESS_NAME_1",
    "business_application_id",
]


COLUMN_MAPPING = {
    "SUPPLIER_DNB_GLOBAL_ULTIMATE_BUSINESS_NAME":
        "supplier_dnb_global_ultimate_business_name",

    "SUPPLIER_NORMALISED_BUSINESS_NAME":
        "supplier_normalised_business_name",

    "SUPPLIER_KEY_TEXT":
        "supplier_key_text",

    "ENGAGEMENT_DETAIL_ENGAGEMENT_ID":
        "engagement_detail_engagement_id",

    "CONTRACTS_DETAIL_CONTRACT_ID":
        "contracts_detail_contract_id",

    "CONTRACTS_DETAIL_CONTRACT_NUMBER":
        "contracts_detail_contract_number",

    "SUPPLIER_DNB_GLOBAL_ULTIMATE_BUSINESS_NAME_1":
        "supplier_dnb_global_ultimate_business_name_1",

    "business_application_id":
        "business_application_id",
}


def _read_excel(
    file_path: str = EXCEL_FILE,
) -> pd.DataFrame:
    """Read the relationship data from Excel."""

    df = pd.read_excel(
        file_path,
        sheet_name=SHEET_NAME,
    )

    return df


def _validate_columns(df: pd.DataFrame) -> None:
    """Validate that all expected source columns exist."""

    missing_columns = [
        column
        for column in SOURCE_COLUMNS
        if column not in df.columns
    ]

    if missing_columns:
        raise ValueError(
            "Missing columns in Excel sheet: "
            + ", ".join(missing_columns)
        )


def _clean_dataframe(df: pd.DataFrame) -> pd.DataFrame:
    """Clean source data before database insertion."""

    df = df[SOURCE_COLUMNS].copy()

    # Convert pandas NaN/NaT values to Python None.
    df = df.where(pd.notna(df), None)

    # Remove exact duplicate source rows.
    before = len(df)

    df = df.drop_duplicates()

    after = len(df)

    if before != after:
        print(
            f"Removed {before - after} exact duplicate rows."
        )

    return df


def _insert_rows(df: pd.DataFrame) -> int:
    """Insert relationship rows into the database."""

    records = []

    for _, row in df.iterrows():

        record = {
            COLUMN_MAPPING[source_column]: row[source_column]
            for source_column in SOURCE_COLUMNS
        }

        records.append(record)

    if not records:
        print("No records found to insert.")
        return 0

    session = Session(engine)

    try:
        session.execute(
            insert(DimTpEngContrBusAppRel),
            records,
        )

        session.commit()

        return len(records)

    except Exception:
        session.rollback()
        raise

    finally:
        session.close()


def main() -> int:
    """Execute the complete Excel-to-database load."""

    print("Reading Excel data...")

    df = _read_excel()

    print(f"Rows read from Excel: {len(df)}")

    _validate_columns(df)

    df = _clean_dataframe(df)

    print(f"Rows after cleaning: {len(df)}")

    inserted_count = _insert_rows(df)

    print(
        "DIM_TP_ENG_CONTR_BUSAPP_REL data loaded successfully."
    )

    print(f"Rows inserted: {inserted_count}")

    return 0


if __name__ == "__main__":
    raise SystemExit(main())
