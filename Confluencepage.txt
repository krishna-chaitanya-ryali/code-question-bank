from sqlalchemy import text
from db import engine

TABLE_NAME = "SCOTT_DETAILS"


def create_table():
    ddl = f"""
    IF OBJECT_ID('{TABLE_NAME}', 'U') IS NULL
    BEGIN
        CREATE TABLE {TABLE_NAME} (
            BUSINESS_APPLICATION_ID VARCHAR(100) NULL,
            BUSINESS_APPLICATION_NAME VARCHAR(500) NULL,
            BUSINESS_APPLICATION_RESILIENCE_RAG_NAME VARCHAR(100) NULL,
            BUSINESS_APPLICATION_RESILIENCE_STATUS_NAME VARCHAR(200) NULL,
            BUSINESS_APPLICATION_DISASTER_RECOVERY_REQUIREMENT_NAME VARCHAR(200) NULL,
            DISASTER_RECOVERY_TEST_LAST_COMPLETION_DATE DATE NULL,
            BUSINESS_APPLICATION_RESILIENCE_RAG_NAME_1 VARCHAR(100) NULL,
            BUSINESS_APPLICATION_RESILIENCE_STATUS_NAME_1 VARCHAR(200) NULL,
            BUSINESS_APPLICATION_LAST_DR_TEST_REVIEW_DATE DATE NULL,
            DISASTER_RECOVERY_TEST_LAST_COMPLETION_DATE_1 DATE NULL,
            DISASTER_RECOVERY_TEST_SCHEDULED_TEST_DATE DATE NULL,
            DISASTER_RECOVERY_TEST_DUE_DATE DATE NULL,
            BUSINESS_APPLICATION_RECOVERY_POINT_OBJECTIVE_NAME VARCHAR(200) NULL,
            BUSINESS_APPLICATION_RECOVERY_TIME_OBJECTIVE_NAME VARCHAR(200) NULL
        );
    END
    """

    with engine.begin() as conn:
        conn.execute(text(ddl))

    print(f"Table {TABLE_NAME} created successfully (or already exists).")


if __name__ == "__main__":
    create_table()

import pandas as pd
from sqlalchemy import text
from db import engine

TABLE_NAME = "SCOTT_DETAILS"
EXCEL_FILE = "Data workbook.xlsx"
SHEET_NAME = "SCOTT_DETAILS"


SOURCE_TO_TARGET = {
    "BUSINESS_APPLICATION_ID": "BUSINESS_APPLICATION_ID",
    "BUSINESS_APPLICATION_NAME": "BUSINESS_APPLICATION_NAME",
    "BUSINESS_APPLICATION_RESILIENCE_RAG_NAME": "BUSINESS_APPLICATION_RESILIENCE_RAG_NAME",
    "BUSINESS_APPLICATION_RESILIENCE_STATUS_NAME": "BUSINESS_APPLICATION_RESILIENCE_STATUS_NAME",
    "BUSINESS_APPLICATION_DISASTER_RECOVERY_REQUIREMENT_NAME": "BUSINESS_APPLICATION_DISASTER_RECOVERY_REQUIREMENT_NAME",
    "DISASTER_RECOVERY_TEST_LAST_COMPLETION_DATE": "DISASTER_RECOVERY_TEST_LAST_COMPLETION_DATE",
    "BUSINESS_APPLICATION_RESILIENCE_RAG_NAME_1": "BUSINESS_APPLICATION_RESILIENCE_RAG_NAME_1",
    "BUSINESS_APPLICATION_RESILIENCE_STATUS_NAME_1": "BUSINESS_APPLICATION_RESILIENCE_STATUS_NAME_1",
    "BUSINESS_APPLICATION_LAST_DR_TEST_REVIEW_DATE": "BUSINESS_APPLICATION_LAST_DR_TEST_REVIEW_DATE",
    "DISASTER_RECOVERY_TEST_LAST_COMPLETION_DATE_1": "DISASTER_RECOVERY_TEST_LAST_COMPLETION_DATE_1",
    "DISASTER_RECOVERY_TEST_SCHEDULED_TEST_DATE": "DISASTER_RECOVERY_TEST_SCHEDULED_TEST_DATE",
    "DISASTER_RECOVERY_TEST_DUE_DATE": "DISASTER_RECOVERY_TEST_DUE_DATE",
    "BUSINESS_APPLICATION_RECOVERY_POINT_OBJECTIVE_NAME": "BUSINESS_APPLICATION_RECOVERY_POINT_OBJECTIVE_NAME",
    "BUSINESS_APPLICATION_RECOVERY_TIME_OBJECTIVE_NAME": "BUSINESS_APPLICATION_RECOVERY_TIME_OBJECTIVE_NAME",
}


def load_data():

    print(f"Reading sheet '{SHEET_NAME}' from '{EXCEL_FILE}'...")

    df = pd.read_excel(
        EXCEL_FILE,
        sheet_name=SHEET_NAME,
        dtype=str
    )

    # Remove accidental spaces from column headers
    df.columns = df.columns.str.strip()

    # Validate expected source columns
    expected_columns = list(SOURCE_TO_TARGET.keys())

    missing_columns = [
        column
        for column in expected_columns
        if column not in df.columns
    ]

    if missing_columns:
        raise ValueError(
            f"Missing expected columns in sheet '{SHEET_NAME}': "
            f"{missing_columns}"
        )

    # Keep only required columns
    df = df[expected_columns]

    # Rename source columns to target database columns
    df = df.rename(columns=SOURCE_TO_TARGET)

    # Convert blank/whitespace values to NULL
    df = df.replace(r"^\s*$", pd.NA, regex=True)

    # Convert date columns
    date_columns = [
        "DISASTER_RECOVERY_TEST_LAST_COMPLETION_DATE",
        "BUSINESS_APPLICATION_LAST_DR_TEST_REVIEW_DATE",
        "DISASTER_RECOVERY_TEST_LAST_COMPLETION_DATE_1",
        "DISASTER_RECOVERY_TEST_SCHEDULED_TEST_DATE",
        "DISASTER_RECOVERY_TEST_DUE_DATE",
    ]

    for column in date_columns:
        df[column] = pd.to_datetime(
            df[column],
            errors="coerce"
        ).dt.date

    print(f"Rows read from Excel: {len(df):,}")

    # Clear existing data before reload
    with engine.begin() as conn:
        conn.execute(text(f"DELETE FROM {TABLE_NAME}"))

    print(f"Existing data cleared from {TABLE_NAME}.")

    # Load data into Azure SQL
    df.to_sql(
        TABLE_NAME,
        con=engine,
        if_exists="append",
        index=False,
        chunksize=500,
        method=None
    )

    print(
        f"Successfully loaded {len(df):,} rows "
        f"into {TABLE_NAME}."
    )


if __name__ == "__main__":
    load_data()
