SELECT
    COUNT(*) AS TOTAL_ROWS,
    COUNT(DISTINCT IMPORTANT_BUSINESS_SERVICE) AS BUSINESS_SERVICE,
    COUNT(DISTINCT ENGAGEMENT_ID) AS ENGAGEMENT,
    COUNT(DISTINCT APP_ID) AS APPLICATION,
    COUNT(DISTINCT ENGAGEMENT_NAME) AS ENGAGEMENT_NAME,
    COUNT(DISTINCT APP_NAME) AS APP_NAME,
    COUNT(DISTINCT SUPPLIER_NAME) AS SUPPLIER,
    COUNT(DISTINCT ENGAGEMENT_MANAGER) AS MANAGER
FROM IMPORTANT_BUSINESS_SERVICE_DETAILS;
