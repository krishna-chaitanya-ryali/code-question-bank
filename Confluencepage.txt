from sqlalchemy import text

from db import engine


TABLE_NAME = "BS4_TECHNOLOGY_THRESHOLDS"


def create_table():
    create_sql = text(f"""
        IF NOT EXISTS (
            SELECT 1
            FROM INFORMATION_SCHEMA.TABLES
            WHERE TABLE_NAME = '{TABLE_NAME}'
        )
        BEGIN
            CREATE TABLE {TABLE_NAME} (
                Incidents INT NULL,
                Outages INT NULL,
                [Service Instances] INT NULL,
                RPO INT NULL,
                Target_availability DECIMAL(5,2) NULL
            );
        END
    """)

    with engine.begin() as connection:
        connection.execute(create_sql)

    print(f"Table '{TABLE_NAME}' created successfully or already exists.")


if __name__ == "__main__":
    create_table()

import pandas as pd

from db import engine


EXCEL_FILE = "Data Workbook.xlsx"
SHEET_NAME = "DBO_Service_Thresholds"
TABLE_NAME = "BS4_TECHNOLOGY_THRESHOLDS"


def load_data():

    required_columns = [
        "Incidents",
        "Outages",
        "Service Instances",
        "RPO",
        "Target_availability"
    ]

    # Read the required columns from Excel
    df = pd.read_excel(
        EXCEL_FILE,
        sheet_name=SHEET_NAME,
        usecols=required_columns
    )

    # Convert numeric columns
    numeric_columns = [
        "Incidents",
        "Outages",
        "Service Instances",
        "RPO",
        "Target_availability"
    ]

    for column in numeric_columns:
        df[column] = pd.to_numeric(
            df[column],
            errors="coerce"
        )

    # Load data into Azure SQL
    df.to_sql(
        TABLE_NAME,
        con=engine,
        if_exists="append",
        index=False
    )

    print(
        f"{len(df)} records loaded successfully "
        f"into '{TABLE_NAME}'."
    )


if __name__ == "__main__":
    load_data()
