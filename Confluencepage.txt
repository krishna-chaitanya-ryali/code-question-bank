SELECT
    COUNT(*) AS NULL_COUNT
FROM BUS_SERV_AWARE_ENGAGEMENT
WHERE ENGAGEMENT_ID IS NULL
   OR BUSINESS_APPLICATION_ID IS NULL;



SELECT
    ENGAGEMENT_ID,
    BUSINESS_APPLICATION_ID,
    COUNT(*) AS RECORD_COUNT
FROM BUS_SERV_AWARE_ENGAGEMENT
GROUP BY
    ENGAGEMENT_ID,
    BUSINESS_APPLICATION_ID
HAVING COUNT(*) > 1;
