"""Load selected engagement data from Excel into Azure SQL.

The script reads the 'Engagement Info' worksheet and loads only the
specified engagement IDs into the ENGAGEMENT table.
"""

import pandas as pd

from sqlalchemy.orm import Session

from db import engine
from models import Engagement


# Engagement IDs required for the current POC
TARGET_ENGAGEMENT_IDS = {
    "EN-021445",
    "ENG-1162865",
    "ENG-362309",
}


def _read_engagement_excel(
    file_path: str = "engagement_info.xlsx",
) -> pd.DataFrame:
    """Read engagement records from the source Excel workbook."""

    return pd.read_excel(
        file_path,
        sheet_name="Engagement Info",
    )


def _filter_target_engagements(df: pd.DataFrame) -> pd.DataFrame:
    """Return only the engagement records required for the current POC."""

    id_column = "ENGAGEMENT_DETAIL_ENGAGEMENT_ID"

    # Convert IDs to string and remove accidental leading/trailing spaces.
    df[id_column] = df[id_column].astype(str).str.strip()

    filtered_df = df[
        df[id_column].isin(TARGET_ENGAGEMENT_IDS)
    ].copy()

    found_ids = set(filtered_df[id_column])

    missing_ids = TARGET_ENGAGEMENT_IDS - found_ids

    print(f"Requested engagement IDs : {len(TARGET_ENGAGEMENT_IDS)}")
    print(f"Found engagement IDs     : {len(found_ids)}")

    if found_ids:
        print("Found IDs:")
        for engagement_id in sorted(found_ids):
            print(f"  - {engagement_id}")

    if missing_ids:
        print("WARNING: The following engagement IDs were not found:")
        for engagement_id in sorted(missing_ids):
            print(f"  - {engagement_id}")

    return filtered_df


def _upsert_engagement_rows(df: pd.DataFrame) -> int:
    """Upsert selected engagement rows into ENGAGEMENT table."""

    session = Session(engine)

    try:
        for _, row in df.iterrows():

            engagement = Engagement(
                engagement_id=row["ENGAGEMENT_DETAIL_ENGAGEMENT_ID"],

                supplier_key_text=row["SUPPLIER_KEY_TEXT"],

                engagement_name=row[
                    "ENGAGEMENT_DETAIL_ENGAGEMENT_NAME"
                ],

                engagement_desc=row[
                    "ENGAGEMENT_DETAIL_DESC"
                ],

                engagement_status=row[
                    "ENGAGEMENT_DETAIL_ENGAGEMENT_STATUS_NAME"
                ],

                category_name=row[
                    "ENGAGEMENT_DETAIL_CATEGORY_NAME"
                ],

                important_third_party_service=row[
                    "IMPORTANT_THIRD_PARTY_SERVICE_IND"
                ],

                important_business_service=row[
                    "ENGAGEMENT_DETAIL_IMPORTANT_BUSINESS_SERVICE_NAME"
                ],

                materiality_assessment=row[
                    "ENGAGEMENT_DETAIL_MATERIALITY_ASSESSMENT_OUTCOME_TEXT"
                ],

                inherent_rating=row[
                    "ENGAGEMENT_DETAIL_INHERENT_RATING_TEXT"
                ],

                application_instance_id=row[
                    "ENGAGEMENT_DETAIL_APPLICATION_INSTANCE_ID"
                ],

                global_business_name=row[
                    "SUPPLIER_DNB_GLOBAL_ULTIMATE_BUSINESS_NAME"
                ],

                type=row["type"],

                tpem_employee_id=row[
                    "TPEM_EMPLOYEE_ID"
                ],

                tpem_email=row[
                    "TPEM_EMAIL_ID"
                ],

                engagement_risk_owner_employee_id=row[
                    "ENGAGEMENT_RISK_OWNER_EMPLOYEE_ID"
                ],

                engagement_risk_owner_email=row[
                    "ENGAGEMENT_RISK_OWNER_EMAIL_ID"
                ],
            )

            # Insert if the ID doesn't exist.
            # Update if the ID already exists.
            session.merge(engagement)

        session.commit()

        return len(df)

    except Exception:
        session.rollback()
        raise

    finally:
        session.close()


def main() -> int:
    """Execute the selected engagement data load."""

    print("Reading engagement Excel file...")

    df = _read_engagement_excel()

    print(f"Total records in Excel: {len(df)}")

    # Keep only the three requested engagement IDs.
    df = _filter_target_engagements(df)

    if df.empty:
        print("No target engagement records found. Nothing to load.")
        return 0

    print(f"\nLoading {len(df)} engagement record(s)...")

    row_count = _upsert_engagement_rows(df)

    print(
        f"\nData loaded successfully. "
        f"Rows processed: {row_count}"
    )

    return 0


if __name__ == "__main__":
    raise SystemExit(main())
