"""Load contract data from Excel into Azure SQL.

The script reads the 'Contract level info' worksheet and loads contracts
for the three in-scope engagement IDs used for the current POC.
"""

import pandas as pd

from sqlalchemy.orm import Session

from db import engine
from models import Contract


# ---------------------------------------------------------
# In-scope engagements for the current POC
# ---------------------------------------------------------

IN_SCOPE_ENGAGEMENT_IDS = {
    "EN-021445",
    "ENG-1162865",
    "ENG-362309",
}


# ---------------------------------------------------------
# Read Excel
# ---------------------------------------------------------

def read_contract_excel(
    file_path: str = "engagement_info.xlsx",
) -> pd.DataFrame:
    """Read contract records from the Contract level info worksheet."""

    return pd.read_excel(
        file_path,
        sheet_name="Contract level info",
    )


# ---------------------------------------------------------
# Clean values
# ---------------------------------------------------------

def clean_value(value):
    """Convert pandas NaN values to None and normalize text."""

    if pd.isna(value):
        return None

    return str(value).strip()


# ---------------------------------------------------------
# Load contracts
# ---------------------------------------------------------

def load_contract_rows(df: pd.DataFrame) -> int:
    """Insert/update contract rows using contract_ref_num as the key."""

    # Keep only contracts belonging to the three in-scope engagements.
    df = df[
        df["ENGAGEMENT_DETAIL_ENGAGEMENT_ID"]
        .astype(str)
        .str.strip()
        .isin(IN_SCOPE_ENGAGEMENT_IDS)
    ].copy()

    print(
        f"Contracts found for in-scope engagements: {len(df)}"
    )

    if df.empty:
        print("No contract records found for the selected engagements.")
        return 0

    session = Session(engine)

    processed_count = 0

    try:

        for _, row in df.iterrows():

            engagement_id = clean_value(
                row["ENGAGEMENT_DETAIL_ENGAGEMENT_ID"]
            )

            supplier_id = clean_value(
                row["SUPPLIER_KEY_TEXT"]
            )

            contract_number = clean_value(
                row["CONTRACTS_DETAIL_CONTRACT_NUMBER"]
            )

            # -------------------------------------------------
            # Basic validation
            # -------------------------------------------------

            if not contract_number:
                print(
                    "Skipping row because CONTRACT_NUMBER is empty."
                )
                continue

            if engagement_id not in IN_SCOPE_ENGAGEMENT_IDS:
                print(
                    f"Skipping contract {contract_number}: "
                    f"engagement {engagement_id} is out of scope."
                )
                continue

            if not supplier_id:
                print(
                    f"Skipping contract {contract_number}: "
                    "SUPPLIER_KEY_TEXT is empty."
                )
                continue

            # -------------------------------------------------
            # Create ORM object
            # -------------------------------------------------

            contract = Contract(
                contract_ref_num=contract_number,

                contract_engagement_id=engagement_id,

                contract_third_party_id=supplier_id,

                contract_id=clean_value(
                    row["CONTRACTS_DETAIL_CONTRACT_ID"]
                ),

                contract_name=clean_value(
                    row["CONTRACTS_DETAIL_CONTRACT_NAME"]
                ),

                contract_desc=clean_value(
                    row["CONTRACTS_DETAIL_CONTRACT_DESC"]
                ),

                regional_coverage=clean_value(
                    row["CONTRACTS_DETAIL_REGIONAL_COVERAGE_TEXT"]
                ),

                contract_status=clean_value(
                    row["CONTRACTS_DETAIL_CONTRACT_STATUS_CODE"]
                ),

                termination_notice=clean_value(
                    row["CONTRACTS_DETAIL_TERMINATION_NOTICE_TEXT"]
                ),

                termination_notice_length=clean_value(
                    row[
                        "CONTRACTS_DETAIL_TERMINATION_NOTICE_LENGTH_TEXT"
                    ]
                ),

                terminated=clean_value(
                    row["CONTRACTS_DETAIL_TERMINATED_TEXT"]
                ),

                termination_reason=clean_value(
                    row["CONTRACTS_DETAIL_TERMINATION_REASON_CODE"]
                ),

                supplier_notice_period_days=clean_value(
                    row[
                        "CONTRACTS_DETAIL_SUPPLIER_NOTICE_PERIOD_DAYS_NUM"
                    ]
                ),

                termination_convenience_right=clean_value(
                    row[
                        "CONTRACTS_DETAIL_TERMINATION_CONVENIENCE_RIGHT_IND"
                    ]
                ),
            )

            # -------------------------------------------------
            # Insert or update
            # -------------------------------------------------

            session.merge(contract)

            processed_count += 1

        session.commit()

        return processed_count

    except Exception:
        session.rollback()
        raise

    finally:
        session.close()


# ---------------------------------------------------------
# Main
# ---------------------------------------------------------

def main() -> int:

    print("Reading contract Excel data...")

    df = read_contract_excel()

    print(
        f"Total rows in Excel sheet: {len(df)}"
    )

    processed_count = load_contract_rows(df)

    print(
        f"Contract data loaded successfully. "
        f"Rows processed: {processed_count}"
    )

    return 0


if __name__ == "__main__":
    raise SystemExit(main())
