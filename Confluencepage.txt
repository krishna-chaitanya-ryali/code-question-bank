from sqlalchemy import text

from db import engine


def alter_table():
    alter_sql = text("""
        ALTER TABLE SERVICE_THRESHOLD_DETAILS
        ADD
            RPO NUMERIC(18, 2) NULL,
            AVAILABILITY_TARGET NUMERIC(5, 2) NULL;
    """)

    with engine.begin() as connection:
        connection.execute(alter_sql)

    print(
        "RPO and AVAILABILITY_TARGET columns added successfully "
        "to SERVICE_THRESHOLD_DETAILS."
    )


if __name__ == "__main__":
    alter_table()
